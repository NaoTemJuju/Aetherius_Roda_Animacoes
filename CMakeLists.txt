cmake_minimum_required(VERSION 3.24)
project(EmotesSP VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set(COMMONLIB_INCLUDE "" CACHE PATH "alandtse/CommonLibVR ng include folder")
set(COMMONLIB_LIBRARY "" CACHE FILEPATH "ABI-matching Release SE+AE CommonLibSSE.lib")
set(DEPENDENCIES "" CACHE PATH "vcpkg x64-windows-static-md install prefix")
if(NOT EXISTS "${COMMONLIB_INCLUDE}/SKSE/SKSE.h" OR NOT EXISTS "${COMMONLIB_LIBRARY}")
  message(FATAL_ERROR "Provide the CommonLib NG headers and matching SE+AE /MD library")
endif()
add_library(EmotesSP SHARED src/plugin.cpp)
target_include_directories(EmotesSP PRIVATE src third_party "${COMMONLIB_INCLUDE}" "${DEPENDENCIES}/include" "${DEPENDENCIES}/include/directxtk")
target_compile_definitions(EmotesSP PRIVATE ENABLE_SKYRIM_SE=1 ENABLE_SKYRIM_AE=1 HAS_SKYRIM_MULTI_TARGETING=1 WINVER=0x0601 _WIN32_WINNT=0x0601 SPDLOG_COMPILED_LIB SPDLOG_FMT_EXTERNAL)
target_compile_options(EmotesSP PRIVATE /Zc:preprocessor /permissive- /utf-8 /EHsc /W4 /wd4200 /wd4324 /wd4100 /wd4005)
target_precompile_headers(EmotesSP PRIVATE src/pch.h)
target_link_libraries(EmotesSP PRIVATE "${COMMONLIB_LIBRARY}" "${DEPENDENCIES}/lib/spdlog.lib" "${DEPENDENCIES}/lib/fmt.lib" "${DEPENDENCIES}/lib/DirectXTK.lib" advapi32 bcrypt d3d11 d3dcompiler dbghelp dxgi ole32 version)
install(TARGETS EmotesSP RUNTIME DESTINATION SKSE/Plugins LIBRARY DESTINATION SKSE/Plugins)
install(DIRECTORY ui/ DESTINATION MeridianUI/emotes-sp)
include(CTest)
if(BUILD_TESTING)
  add_executable(EmotesSPTests tests/session.cpp)
  target_include_directories(EmotesSPTests PRIVATE src third_party "${DEPENDENCIES}/include")
  target_compile_options(EmotesSPTests PRIVATE /utf-8 /EHsc)
  add_test(NAME meridian-session COMMAND EmotesSPTests)
endif()
